select 	a.name, max(backup_finish_date) as bkup_finish_date
INTO 	#mx
from 	master.sys.databases a
LEFT 	OUTER JOIN msdb..backupset b
ON      a.name = b.database_name
where	a.name <> 'tempdb' and a.state_desc not in('OFFLINE','RESTORING') and a.is_read_only=0 and a.name not like '%mock%' and a.name not in
('XXXXX',
'XXXX',
'XXXXXXXXX',
'XXXXXXXXXXXXXXXXXXXXXXXXXXX')
and b.type = 'D'
and a.database_id not in (select database_id from sys.database_mirroring where mirroring_guid is not null and mirroring_role_desc = 'MIRROR')
--or b.type ='I'
group	by a.name
order	by a.name



select 	a.name as dbname ,
	CONVERT(decimal(20,2),Round((backup_size/1048576),0)) as size_mb, 	
			CASE WHEN b.backup_start_date > DATEADD(dd,-8,getdate())
--convert(int,b.backup_start_date,101) >= convert(int,DATEADD(dd,-1,getdate()),101)
		  THEN 'Backup Completed'
	     ELSE 'Backup Not Completed'
	END
	AS 'Status',	
	a.bkup_finish_date as rundate
INTO	#backup1
from 	#mx a,
	msdb..backupset b
where 	a.name = b.database_name
and	a.bkup_finish_date = b.backup_finish_date 
and a.name not in ('DBAdmin',
'XXXX',
'XXXXXXXXX',
'XXXXXXXXXXXXXXXXXXXXXXXXXXX')
--and	server_name = CONVERT(varchar(100), serverproperty('servername'))
and b.type = 'D'


select b.name as dbname, coalesce (a.size_mb,0) as size_mb ,coalesce(a.Status,'Backup Not Completed') as [Status] ,coalesce(a.rundate,'') as rundate
into #backup
from #backup1 a
right outer join master.sys.databases b
on a.dbname = b.name
where b.name not in ('DBAdmin',
'XXXX',
'XXXXXXXXX',
'XXXXXXXXXXXXXXXXXXXXXXXXXXX','tempdb')
and a.dbname not in (select db_name(database_id) from sys.database_mirroring where mirroring_guid is not null and mirroring_role_desc = 'MIRROR')
and b.state_desc not in('OFFLINE','RESTORING')

DECLARE bkpup CURSOR FOR SELECT dbname, size_mb, Status, rundate from #backup where dbname not in('XXXXX',
'XXXX',
'XXXXXXXXX',
'XXXXXXXXXXXXXXXXXXXXXXXXXXX')
OPEN bkpup 

FETCH NEXT FROM bkpup
INTO @dbname, @size,    @status, @rundate

WHILE @@FETCH_STATUS = 0
BEGIN
	
	PRINT  '<td align=center><font face=verdana size=1>' +  @dbname + '</font></td>'
	PRINT  '<td align=center><font face=verdana size=1>' +  @size + '</font></td>'
		IF @status <> 'Backup Completed'

	PRINT  '<td align=center><font face=verdana size=1>' +  @rundate + '</font></td></tr>'
	
   FETCH NEXT FROM bkpup
   INTO @dbname, @size, @status, @rundate

END
Close bkpup
